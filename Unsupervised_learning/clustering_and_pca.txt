import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

from google.colab import files

from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans, AgglomerativeClustering, DBSCAN
from sklearn.decomposition import PCA
from sklearn.metrics import silhouette_score
from sklearn.neighbors import NearestNeighbors

from scipy.cluster.hierarchy import dendrogram, linkage

import warnings
warnings.filterwarnings("ignore")


# ---------------------------
# 2. UPLOAD DATASET
# ---------------------------

uploaded = files.upload()

file_name = list(uploaded.keys())[0]

print("\nUploaded File:", file_name)


# ---------------------------
# 3. READ EXCEL FILE
# ---------------------------

try:
    xls = pd.ExcelFile(file_name, engine="openpyxl")
except:
    xls = pd.ExcelFile(file_name, engine="xlrd")

print("\nSheet Names:")
print(xls.sheet_names)


# ---------------------------
# 4. LOAD AND COMBINE
#    ALL SHEETS
# ---------------------------

dataframes = []

for sheet in xls.sheet_names:

    temp_df = pd.read_excel(
        file_name,
        sheet_name=sheet
    )

    dataframes.append(temp_df)


df = pd.concat(
    dataframes,
    ignore_index=True
)

print("\nOriginal Dataset Shape:", df.shape)

print("\nColumn Names:")
print(df.columns.tolist())


# ---------------------------
# 5. CLEAN COLUMN NAMES
# ---------------------------

df.columns = (
    df.columns
    .astype(str)
    .str.strip()
    .str.replace(" ", "_")
)

print("\nCleaned Column Names:")
print(df.columns.tolist())


# ---------------------------
# 6. DATA CLEANING
# ---------------------------

df_clean = df.copy()

# Remove missing Customer ID
df_clean = df_clean.dropna(
    subset=["Customer_ID"]
)

# Convert Invoice to string
df_clean["Invoice"] = (
    df_clean["Invoice"]
    .astype(str)
)

# Remove cancelled invoices
df_clean = df_clean[
    ~df_clean["Invoice"]
    .str.startswith("C")
]

# Keep positive Quantity and Price
df_clean = df_clean[
    (df_clean["Quantity"] > 0) &
    (df_clean["Price"] > 0)
]

# Remove duplicates
df_clean = df_clean.drop_duplicates()

# Convert Invoice Date
df_clean["InvoiceDate"] = pd.to_datetime(
    df_clean["InvoiceDate"],
    errors="coerce"
)

# Remove missing dates
df_clean = df_clean.dropna(
    subset=["InvoiceDate"]
)

print("\nCleaned Dataset Shape:", df_clean.shape)


# ---------------------------
# 7. FEATURE ENGINEERING
# ---------------------------

# Create Total Amount
df_clean["TotalAmount"] = (
    df_clean["Quantity"] *
    df_clean["Price"]
)


# ---------------------------
# 8. CREATE RFM FEATURES
# ---------------------------

reference_date = (
    df_clean["InvoiceDate"].max()
    + pd.Timedelta(days=1)
)

rfm = df_clean.groupby(
    "Customer_ID"
).agg(

    Recency=(
        "InvoiceDate",
        lambda x: (
            reference_date - x.max()
        ).days
    ),

    Frequency=(
        "Invoice",
        "nunique"
    ),

    Monetary=(
        "TotalAmount",
        "sum"
    )
)

print("\nRFM Dataset Shape:", rfm.shape)

print("\nRFM Dataset:")
print(rfm.head())


# ---------------------------
# 9. LOG TRANSFORMATION
# ---------------------------

rfm_log = np.log1p(rfm)


# ---------------------------
# 10. STANDARDIZATION
# ---------------------------

scaler = StandardScaler()

rfm_scaled = scaler.fit_transform(
    rfm_log
)

print("\nScaled Dataset Shape:", rfm_scaled.shape)


# ============================================================
#                    K-MEANS CLUSTERING
# ============================================================

print("\n" + "=" * 60)
print("K-MEANS CLUSTERING")
print("=" * 60)


# ---------------------------
# FIND BEST K
# ---------------------------

inertia = []
silhouette_scores = []

k_range = range(2, 11)

for k in k_range:

    kmeans_temp = KMeans(
        n_clusters=k,
        random_state=42,
        n_init=10
    )

    labels_temp = kmeans_temp.fit_predict(
        rfm_scaled
    )

    inertia.append(
        kmeans_temp.inertia_
    )

    score = silhouette_score(
        rfm_scaled,
        labels_temp
    )

    silhouette_scores.append(score)

    print(
        f"K = {k} | "
        f"Silhouette Score = {score:.4f}"
    )


# ---------------------------
# ELBOW GRAPH
# ---------------------------

plt.figure(figsize=(8, 5))

plt.plot(
    list(k_range),
    inertia,
    marker="o"
)

plt.xlabel("Number of Clusters (K)")
plt.ylabel("Inertia")
plt.title("Elbow Method for K-Means")

plt.show()


# ---------------------------
# SILHOUETTE GRAPH
# ---------------------------

plt.figure(figsize=(8, 5))

plt.plot(
    list(k_range),
    silhouette_scores,
    marker="o"
)

plt.xlabel("Number of Clusters (K)")
plt.ylabel("Silhouette Score")
plt.title("Silhouette Score for K-Means")

plt.show()


# ---------------------------
# AUTOMATICALLY SELECT
# BEST K
# ---------------------------

best_k = list(k_range)[
    np.argmax(silhouette_scores)
]

print("\nBest K based on Silhouette Score:", best_k)


# ---------------------------
# TRAIN FINAL K-MEANS
# ---------------------------

kmeans = KMeans(
    n_clusters=best_k,
    random_state=42,
    n_init=10
)

kmeans_labels = kmeans.fit_predict(
    rfm_scaled
)

rfm["KMeans_Cluster"] = (
    kmeans_labels
)

kmeans_score = silhouette_score(
    rfm_scaled,
    kmeans_labels
)

print(
    "\nFinal K-Means Silhouette Score:",
    kmeans_score
)

print("\nK-Means Cluster Counts:")
print(
    rfm["KMeans_Cluster"]
    .value_counts()
    .sort_index()
)


# ============================================================
#                         PCA
# ============================================================

print("\n" + "=" * 60)
print("PCA")
print("=" * 60)


pca = PCA(
    n_components=2
)

pca_data = pca.fit_transform(
    rfm_scaled
)

print(
    "\nExplained Variance Ratio:",
    pca.explained_variance_ratio_
)

print(
    "Total Explained Variance:",
    pca.explained_variance_ratio_.sum()
)


# Create PCA DataFrame
pca_df = pd.DataFrame(
    pca_data,
    columns=["PC1", "PC2"]
)

pca_df["KMeans_Cluster"] = (
    kmeans_labels
)


# PCA Visualization
plt.figure(figsize=(10, 6))

plt.scatter(
    pca_df["PC1"],
    pca_df["PC2"],
    c=pca_df["KMeans_Cluster"],
    alpha=0.6
)

plt.xlabel("Principal Component 1")
plt.ylabel("Principal Component 2")

plt.title(
    "K-Means Clusters Visualized using PCA"
)

plt.show()


# ============================================================
#                HIERARCHICAL CLUSTERING
# ============================================================

print("\n" + "=" * 60)
print("HIERARCHICAL CLUSTERING")
print("=" * 60)


# Hierarchical clustering is expensive
# So use a maximum of 3000 samples

sample_size = min(
    3000,
    len(rfm_scaled)
)

np.random.seed(42)

sample_indices = np.random.choice(
    len(rfm_scaled),
    sample_size,
    replace=False
)

sample_data = rfm_scaled[
    sample_indices
]

print(
    "\nHierarchical Sample Shape:",
    sample_data.shape
)


# ---------------------------
# DENDROGRAM
# ---------------------------

linked = linkage(
    sample_data,
    method="ward"
)

plt.figure(figsize=(15, 7))

dendrogram(
    linked,
    truncate_mode="lastp",
    p=30
)

plt.title(
    "Hierarchical Clustering Dendrogram"
)

plt.xlabel("Customer Groups")
plt.ylabel("Distance")

plt.show()


# ---------------------------
# TRAIN HIERARCHICAL MODEL
# ---------------------------

hierarchical_clusters = best_k

hierarchical = AgglomerativeClustering(
    n_clusters=hierarchical_clusters,
    linkage="ward"
)

hierarchical_labels = (
    hierarchical.fit_predict(
        sample_data
    )
)

hierarchical_score = silhouette_score(
    sample_data,
    hierarchical_labels
)

print(
    "\nHierarchical Silhouette Score:",
    hierarchical_score
)


# ---------------------------
# HIERARCHICAL PCA
# ---------------------------

hierarchical_pca = PCA(
    n_components=2
)

hierarchical_pca_data = (
    hierarchical_pca.fit_transform(
        sample_data
    )
)

plt.figure(figsize=(10, 6))

plt.scatter(
    hierarchical_pca_data[:, 0],
    hierarchical_pca_data[:, 1],
    c=hierarchical_labels,
    alpha=0.6
)

plt.xlabel("PC1")
plt.ylabel("PC2")

plt.title(
    "Hierarchical Clustering Visualized using PCA"
)

plt.show()


# ============================================================
#                      DBSCAN
# ============================================================

print("\n" + "=" * 60)
print("DBSCAN CLUSTERING")
print("=" * 60)


# ---------------------------
# FIND EPS USING
# K-DISTANCE GRAPH
# ---------------------------

min_samples = 5

neighbors = NearestNeighbors(
    n_neighbors=min_samples
)

neighbors.fit(
    sample_data
)

distances, indices = (
    neighbors.kneighbors(
        sample_data
    )
)

distances = np.sort(
    distances[:, min_samples - 1]
)


plt.figure(figsize=(10, 5))

plt.plot(distances)

plt.xlabel("Data Points")
plt.ylabel("5th Nearest Neighbor Distance")

plt.title(
    "K-Distance Graph for DBSCAN"
)

plt.show()


# ---------------------------
# TEST MULTIPLE EPS VALUES
# AND AUTOMATICALLY SELECT BEST
# ---------------------------

eps_values = np.arange(
    0.1,
    2.1,
    0.1
)

best_eps = None
best_dbscan_score = -1
best_dbscan_labels = None


for eps in eps_values:

    dbscan_temp = DBSCAN(
        eps=eps,
        min_samples=min_samples
    )

    temp_labels = (
        dbscan_temp.fit_predict(
            sample_data
        )
    )

    # Ignore noise points
    mask = temp_labels != -1

    unique_clusters = np.unique(
        temp_labels[mask]
    )

    # Calculate score only
    # if more than 1 cluster exists
    if len(unique_clusters) > 1:

        score = silhouette_score(
            sample_data[mask],
            temp_labels[mask]
        )

        print(
            f"EPS = {eps:.1f} | "
            f"Clusters = {len(unique_clusters)} | "
            f"Silhouette Score = {score:.4f}"
        )

        if score > best_dbscan_score:

            best_dbscan_score = score
            best_eps = eps
            best_dbscan_labels = (
                temp_labels
            )


# ---------------------------
# FINAL DBSCAN MODEL
# ---------------------------

if best_eps is not None:

    dbscan = DBSCAN(
        eps=best_eps,
        min_samples=min_samples
    )

    dbscan_labels = dbscan.fit_predict(
        sample_data
    )

    n_clusters = (
        len(set(dbscan_labels))
        - (1 if -1 in dbscan_labels else 0)
    )

    n_noise = list(
        dbscan_labels
    ).count(-1)

    dbscan_score = best_dbscan_score

    print(
        "\nBest EPS:",
        best_eps
    )

    print(
        "Number of DBSCAN Clusters:",
        n_clusters
    )

    print(
        "Number of Noise Points:",
        n_noise
    )

    print(
        "Best DBSCAN Silhouette Score:",
        dbscan_score
    )


    # DBSCAN PCA VISUALIZATION

    dbscan_pca = PCA(
        n_components=2
    )

    dbscan_pca_data = (
        dbscan_pca.fit_transform(
            sample_data
        )
    )

    plt.figure(figsize=(10, 6))

    plt.scatter(
        dbscan_pca_data[:, 0],
        dbscan_pca_data[:, 1],
        c=dbscan_labels,
        alpha=0.6
    )

    plt.xlabel("PC1")
    plt.ylabel("PC2")

    plt.title(
        "DBSCAN Clustering Visualized using PCA"
    )

    plt.show()

else:

    print(
        "\nDBSCAN could not find "
        "more than one valid cluster."
    )

    dbscan_score = np.nan


# ============================================================
#                  FINAL COMPARISON
# ============================================================

print("\n" + "=" * 60)
print("FINAL MODEL COMPARISON")
print("=" * 60)


results = pd.DataFrame({

    "Algorithm": [
        "K-Means",
        "Hierarchical",
        "DBSCAN"
    ],

    "Silhouette Score": [
        kmeans_score,
        hierarchical_score,
        dbscan_score
    ]
})


print("\nModel Comparison:")

print(
    results.sort_values(
        by="Silhouette Score",
        ascending=False
    )
)


# ============================================================
#                CLUSTER ANALYSIS
# ============================================================

print("\n" + "=" * 60)
print("K-MEANS CUSTOMER CLUSTER ANALYSIS")
print("=" * 60)


cluster_analysis = rfm.groupby(
    "KMeans_Cluster"
).agg(

    Recency_Mean=(
        "Recency",
        "mean"
    ),

    Frequency_Mean=(
        "Frequency",
        "mean"
    ),

    Monetary_Mean=(
        "Monetary",
        "mean"
    ),

    Customer_Count=(
        "Recency",
        "count"
    )
)


print(cluster_analysis)


# ============================================================
#                    COMPLETED
# ============================================================

print("\n" + "=" * 60)
print("ALL ALGORITHMS COMPLETED SUCCESSFULLY")
print("=" * 60)

print("\nAlgorithms Used:")
print("1. K-Means Clustering")
print("2. Hierarchical Clustering")
print("3. DBSCAN Clustering")
print("4. PCA - Principal Component Analysis")
