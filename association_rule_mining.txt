import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import time

from google.colab import files

from mlxtend.preprocessing import TransactionEncoder
from mlxtend.frequent_patterns import (
    apriori,
    fpgrowth,
    association_rules
)


# ============================================================
# 3. UPLOAD DATASET
# ============================================================

uploaded = files.upload()

file_name = list(uploaded.keys())[0]

print("Uploaded File:", file_name)


# ============================================================
# 4. LOAD DATASET
# ============================================================

df = pd.read_csv(
    file_name,
    header=None
)

print("\nDataset Shape:", df.shape)

print("\nFirst 5 Rows:")
print(df.head())


# ============================================================
# 5. CONVERT DATA INTO TRANSACTIONS
# ============================================================

transactions = []

for i in range(len(df)):
    
    transaction = []
    
    for item in df.iloc[i]:
        
        if pd.notna(item):
            
            item = str(item).strip()
            
            if item != "":
                transaction.append(item)
    
    transactions.append(transaction)


print("\nTotal Transactions:", len(transactions))

print("\nFirst 5 Transactions:")

for transaction in transactions[:5]:
    print(transaction)


# ============================================================
# 6. ONE-HOT ENCODING
# ============================================================

te = TransactionEncoder()

te_array = te.fit(
    transactions
).transform(
    transactions
)

transaction_df = pd.DataFrame(
    te_array,
    columns=te.columns_
)

print("\nTransaction Matrix Shape:")
print(transaction_df.shape)

print("\nTransaction Matrix:")
print(transaction_df.head())


# ============================================================
#                      APRIORI
# ============================================================

print("\n" + "=" * 60)
print("APR IORI ALGORITHM")
print("=" * 60)


start_time = time.time()


apriori_itemsets = apriori(
    transaction_df,
    min_support=0.003,
    use_colnames=True
)


apriori_time = time.time() - start_time


print("\nApriori Execution Time:", round(apriori_time, 4), "seconds")

print("\nApriori Frequent Itemsets:")

print(
    apriori_itemsets
    .sort_values(
        by="support",
        ascending=False
    )
    .head(20)
)


# ------------------------------------------------------------
# APRIORI ASSOCIATION RULES
# ------------------------------------------------------------

apriori_rules = association_rules(
    apriori_itemsets,
    metric="lift",
    min_threshold=1
)


apriori_rules = apriori_rules[
    apriori_rules["confidence"] >= 0.2
]


print("\nTop Apriori Association Rules:")

print(
    apriori_rules[
        [
            "antecedents",
            "consequents",
            "support",
            "confidence",
            "lift"
        ]
    ]
    .sort_values(
        by="lift",
        ascending=False
    )
    .head(20)
)


# ============================================================
#                     FP-GROWTH
# ============================================================

print("\n" + "=" * 60)
print("FP-GROWTH ALGORITHM")
print("=" * 60)


start_time = time.time()


fp_itemsets = fpgrowth(
    transaction_df,
    min_support=0.003,
    use_colnames=True
)


fp_time = time.time() - start_time


print("\nFP-Growth Execution Time:", round(fp_time, 4), "seconds")

print("\nFP-Growth Frequent Itemsets:")

print(
    fp_itemsets
    .sort_values(
        by="support",
        ascending=False
    )
    .head(20)
)


# ------------------------------------------------------------
# FP-GROWTH ASSOCIATION RULES
# ------------------------------------------------------------

fp_rules = association_rules(
    fp_itemsets,
    metric="lift",
    min_threshold=1
)


fp_rules = fp_rules[
    fp_rules["confidence"] >= 0.2
]


print("\nTop FP-Growth Association Rules:")

print(
    fp_rules[
        [
            "antecedents",
            "consequents",
            "support",
            "confidence",
            "lift"
        ]
    ]
    .sort_values(
        by="lift",
        ascending=False
    )
    .head(20)
)


# ============================================================
#                        ECLAT
# ============================================================

print("\n" + "=" * 60)
print("ECLAT ALGORITHM")
print("=" * 60)


# ------------------------------------------------------------
# CREATE VERTICAL DATABASE
# ------------------------------------------------------------

vertical_database = {}

for transaction_id, transaction in enumerate(transactions):
    
    for item in transaction:
        
        if item not in vertical_database:
            
            vertical_database[item] = set()
        
        vertical_database[item].add(
            transaction_id
        )


# ------------------------------------------------------------
# SET MINIMUM SUPPORT
# ------------------------------------------------------------

min_support = 0.003

min_support_count = int(
    min_support * len(transactions)
)


print(
    "\nMinimum Support Count:",
    min_support_count
)


# ------------------------------------------------------------
# ECLAT FUNCTION
# ------------------------------------------------------------

def eclat(
    prefix,
    items,
    frequent_itemsets,
    min_support_count
):
    
    while items:
        
        item, tidset = items.pop()
        
        new_itemset = prefix + [item]
        
        support_count = len(tidset)
        
        
        if support_count >= min_support_count:
            
            frequent_itemsets.append(
                (
                    new_itemset,
                    support_count
                )
            )
            
            
            suffix_items = []
            
            
            for other_item, other_tidset in items:
                
                intersection = (
                    tidset &
                    other_tidset
                )
                
                
                if (
                    len(intersection)
                    >= min_support_count
                ):
                    
                    suffix_items.append(
                        (
                            other_item,
                            intersection
                        )
                    )
            
            
            eclat(
                new_itemset,
                suffix_items,
                frequent_itemsets,
                min_support_count
            )


# ------------------------------------------------------------
# RUN ECLAT
# ------------------------------------------------------------

start_time = time.time()


initial_items = list(
    vertical_database.items()
)


frequent_itemsets_eclat = []


eclat(
    [],
    initial_items,
    frequent_itemsets_eclat,
    min_support_count
)


eclat_time = time.time() - start_time


print(
    "\nECLAT Execution Time:",
    round(eclat_time, 4),
    "seconds"
)


# ------------------------------------------------------------
# CONVERT ECLAT RESULTS TO DATAFRAME
# ------------------------------------------------------------

eclat_results = []


for itemset, support_count in frequent_itemsets_eclat:
    
    eclat_results.append(
        {
            "Itemset": itemset,
            
            "Support_Count": support_count,
            
            "Support": (
                support_count /
                len(transactions)
            )
        }
    )


eclat_df = pd.DataFrame(
    eclat_results
)


print("\nTop ECLAT Frequent Itemsets:")

print(
    eclat_df
    .sort_values(
        by="Support",
        ascending=False
    )
    .head(20)
)


# ============================================================
#               ALGORITHM PERFORMANCE COMPARISON
# ============================================================

print("\n" + "=" * 60)
print("ALGORITHM COMPARISON")
print("=" * 60)


comparison = pd.DataFrame(
    {
        "Algorithm": [
            "Apriori",
            "FP-Growth",
            "ECLAT"
        ],
        
        "Execution Time (seconds)": [
            apriori_time,
            fp_time,
            eclat_time
        ],
        
        "Frequent Itemsets Found": [
            len(apriori_itemsets),
            len(fp_itemsets),
            len(eclat_df)
        ]
    }
)


print("\nPerformance Comparison:")

print(
    comparison
    .sort_values(
        by="Execution Time (seconds)"
    )
)


# ============================================================
#                  EXECUTION TIME GRAPH
# ============================================================

plt.figure(figsize=(8, 5))

plt.bar(
    comparison["Algorithm"],
    comparison["Execution Time (seconds)"]
)

plt.xlabel("Algorithm")

plt.ylabel("Execution Time (seconds)")

plt.title(
    "Execution Time Comparison"
)

plt.show()


# ============================================================
#               FREQUENT ITEMSETS GRAPH
# ============================================================

plt.figure(figsize=(8, 5))

plt.bar(
    comparison["Algorithm"],
    comparison["Frequent Itemsets Found"]
)

plt.xlabel("Algorithm")

plt.ylabel("Number of Frequent Itemsets")

plt.title(
    "Frequent Itemsets Comparison"
)

plt.show()


# ============================================================
#                       FINAL OUTPUT
# ============================================================

print("\n" + "=" * 60)
print("ALL ALGORITHMS COMPLETED SUCCESSFULLY")
print("=" * 60)

print("\nAlgorithms Used:")

print("1. Apriori Algorithm")
print("2. FP-Growth Algorithm")
print("3. ECLAT Algorithm")

print("\nBest Algorithm Based on Execution Time:")

best_algorithm = comparison.loc[
    comparison[
        "Execution Time (seconds)"
    ].idxmin(),
    "Algorithm"
]

print(best_algorithm)
